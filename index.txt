    <script>
        // Banque de questions familiale (adaptée jeunes et moins jeunes)
        const quizData = [

                fact: "Hakuna Matata ! Timon et Pumbaa sont inséparables."
            },
            {

                correct: 1,

            },
            {

            },

            },

                correct: 2,

            },
            {
                question: "Quel animal marin géant est le plus grand animal vivant connu sur Terre ?",
                options: ["Le grand requin blanc", "La baleine bleue", "Le calmar géant", "L'orque"],
                correct: 1,
                fact: "La langue d'une baleine bleue peut peser autant qu'un éléphant adulte !"
            },
            {
                question: "Dans l'univers des super-héros Marvel, quel métal précieux compose le bouclier de Captain America ?",
                options: ["Le titane", "L'adamantium", "Le vibranium", "L'acier pur"],
                correct: 2,
                fact: "Le vibranium absorbe les vibrations cinétiques, le rendant quasi indestructible."
            },
            {
                question: "Quel monument parisien très célèbre a été construit pour l'Exposition universelle de 1889 ?",
                options: ["L'Arc de Triomphe", "La Tour Eiffel", "Le Musée du Louvre", "La cathédrale Notre-Dame"],
                correct: 1,
                fact: "Conçue par Gustave Eiffel, elle mesure initialement 312 mètres de haut."
            }
        ];

        let currentQuestionIndex = 0;
        let score = 0;
        let timer = null;
        let timeLeft = 30;
        let soundEnabled = true;
        let isAnswerLocked = false;
        let audioCtx = null;

        function initAudio() {
            if (!audioCtx) {
                audioCtx = new (window.AudioContext || window.webkitAudioContext)();
            }
            if (audioCtx.state === 'suspended') {
                audioCtx.resume();
            }
        }

        function playSound(type) {
            if (!soundEnabled) return;
            try {
                initAudio();
                const osc = audioCtx.createOscillator();
                const gainNode = audioCtx.createGain();
                osc.connect(gainNode);
                gainNode.connect(audioCtx.destination);

                const now = audioCtx.currentTime;

                if (type === 'correct') {
                    osc.type = 'triangle';
                    osc.frequency.setValueAtTime(440, now);
                    osc.frequency.setValueAtTime(554.37, now + 0.1);
                    osc.frequency.setValueAtTime(659.25, now + 0.2);
                    gainNode.gain.setValueAtTime(0.3, now);
                    gainNode.gain.exponentialRampToValueAtTime(0.01, now + 0.4);
                    osc.start(now);
                    osc.stop(now + 0.4);
                } else if (type === 'wrong') {
                    osc.type = 'sawtooth';
                    osc.frequency.setValueAtTime(220, now);
                    osc.frequency.setValueAtTime(150, now + 0.15);
                    gainNode.gain.setValueAtTime(0.3, now);
                    gainNode.gain.exponentialRampToValueAtTime(0.01, now + 0.3);
                    osc.start(now);
                    osc.stop(now + 0.3);
                } else if (type === 'click') {
                    osc.type = 'sine';
                    osc.frequency.setValueAtTime(600, now);
                    gainNode.gain.setValueAtTime(0.1, now);
                    gainNode.gain.exponentialRampToValueAtTime(0.01, now + 0.05);
                    osc.start(now);
                    osc.stop(now + 0.05);
                } else if (type === 'timeout') {
                    osc.type = 'square';
                    osc.frequency.setValueAtTime(300, now);
                    osc.frequency.setValueAtTime(200, now + 0.2);
                    gainNode.gain.setValueAtTime(0.2, now);
                    gainNode.gain.exponentialRampToValueAtTime(0.01, now + 0.4);
                    osc.start(now);
                    osc.stop(now + 0.4);
                }
            } catch (e) {
                console.log("Audio non disponible", e);
            }
        }

        function speak(text) {
            if (!soundEnabled || !('speechSynthesis' in window)) return;
            try {
                window.speechSynthesis.cancel();
                const utterance = new SpeechSynthesisUtterance(text);
                utterance.lang = 'fr-FR';
                utterance.rate = 1.05;
                utterance.pitch = 1.1;
                window.speechSynthesis.speak(utterance);
            } catch (e) {
                console.log("Synthèse vocale non disponible", e);
            }
        }

        function toggleSound() {
            soundEnabled = !soundEnabled;
            const btn = document.getElementById('soundToggle');
            btn.textContent = soundEnabled ? '🔊' : '🔇';
            if (soundEnabled) speak("Sons activés");
        }

        function startGame() {
            initAudio();
            playSound('click');
            currentQuestionIndex = 0;
            score = 0;
            document.getElementById('scoreDisplay').textContent = score;
            document.getElementById('startScreen').classList.add('hidden');
            document.getElementById('resultScreen').classList.add('hidden');
            document.getElementById('gameScreen').classList.remove('hidden');
            document.getElementById('gameScreen').style.display = 'flex';
            
            speak("C'est parti ! Bonne chance à tous.");
            loadQuestion();
        }

        function loadQuestion() {
            isAnswerLocked = false;
            clearInterval(timer);
            timeLeft = 30;

            const q = quizData[currentQuestionIndex];
            document.getElementById('questionCounter').textContent = `Question ${currentQuestionIndex + 1} / ${quizData.length}`;
            document.getElementById('questionText').textContent = q.question;
            
            speak(q.question);

            const optionsContainer = document.getElementById('optionsContainer');
            optionsContainer.innerHTML = '';

            q.options.forEach((opt, index) => {
                const btn = document.createElement('button');
                btn.className = 'option-btn w-full p-4 bg-white/10 hover:bg-white/20 border border-white/20 rounded-2xl font-semibold text-left sm:text-base flex justify-between items-center group cursor-pointer';
                btn.innerHTML = `<span>${opt}</span> <span class="w-6 h-6 rounded-full bg-white/10 group-hover:bg-white/30 flex items-center justify-center text-xs text-indigo-200">➔</span>`;
                btn.onclick = () => selectAnswer(index);
                optionsContainer.appendChild(btn);
            });

            startTimer();
        }

        function startTimer() {
            const timerText = document.getElementById('timerText');
            const timerBar = document.getElementById('timerBar');
            
            timerBar.style.width = '100%';
            timerText.textContent = `${timeLeft}s`;

            timer = setInterval(() => {
                timeLeft--;
                timerText.textContent = `${timeLeft}s`;
                const percentage = (timeLeft / 30) * 100;
                timerBar.style.width = `${percentage}%`;

                if (timeLeft <= 5 && timeLeft > 0) {
                    playSound('click');
                }

                if (timeLeft <= 0) {
                    clearInterval(timer);
                    handleTimeOut();
                }
            }, 1000);
        }

        function handleTimeOut() {
            if (isAnswerLocked) return;
            isAnswerLocked = true;
            playSound('timeout');
            speak("Temps écoulé !");
            highlightCorrectAnswer();
            
            setTimeout(() => {
                nextQuestion();
            }, 2500);
        }

        function selectAnswer(selectedIndex) {
            if (isAnswerLocked) return;
            isAnswerLocked = true;
            clearInterval(timer);
            playSound('click');

            const q = quizData[currentQuestionIndex];
            const buttons = document.getElementById('optionsContainer').children;

            if (selectedIndex === q.correct) {
                buttons[selectedIndex].className = 'option-btn w-full p-4 bg-green-500/80 border border-green-300 text-white rounded-2xl font-semibold text-left flex justify-between items-center shadow-lg';
                score += 10 + Math.floor(timeLeft / 3);
                document.getElementById('scoreDisplay').textContent = score;
                playSound('correct');
                
                const encouragements = ["Bravo !", "Super !", "Excellent !", "Magnifique !"];
                const randomEnc = encouragements[Math.floor(Math.random() * encouragements.length)];
                speak(`${randomEnc} ${q.fact}`);
                
                if (typeof confetti === 'function') {
                    confetti({
                        particleCount: 50,
                        spread: 60,
                        origin: { y: 0.7 }
                    });
                }
            } else {
                buttons[selectedIndex].className = 'option-btn w-full p-4 bg-red-500/80 border border-red-300 text-white rounded-2xl font-semibold text-left flex justify-between items-center shadow-lg';
                buttons[q.correct].className = 'option-btn w-full p-4 bg-green-500/80 border border-green-300 text-white rounded-2xl font-semibold text-left flex justify-between items-center shadow-lg';
                playSound('wrong');
                speak(`Dommage. La bonne réponse était : ${q.options[q.correct]}`);
            }

            setTimeout(() => {
                nextQuestion();
            }, 3500);
        }

        function highlightCorrectAnswer() {
            const q = quizData[currentQuestionIndex];
            const buttons = document.getElementById('optionsContainer').children;
            if (buttons[q.correct]) {
                buttons[q.correct].className = 'option-btn w-full p-4 bg-green-500/80 border border-green-300 text-white rounded-2xl font-semibold text-left flex justify-between items-center shadow-lg';
            }
        }

        function nextQuestion() {
            currentQuestionIndex++;
            if (currentQuestionIndex < quizData.length) {
                loadQuestion();
            } else {
                endGame();
            }
        }

        function endGame() {
            document.getElementById('gameScreen').classList.add('hidden');
            document.getElementById('resultScreen').classList.remove('hidden');
            document.getElementById('resultScreen').style.display = 'flex';

            const finalScoreEl = document.getElementById('finalScore');
            finalScoreEl.textContent = `${score} pts`;

            let badge = "🌟 Explorateur Curieux";
            let message = "Belle performance ! Continue d'apprendre et de t'amuser en famille !";
            let emoji = "🎉";

            const percentage = (score / (quizData.length * 10)) * 100;

            if (percentage >= 90) {
                badge = "👑 Maître de la Culture Familiale";
                message = "Incroyable ! Tu connais tout sur tout, c'est un sans-faute ou presque !";
                emoji = "🏆";
            } else if (percentage >= 60) {
                badge = "🥇 Expert Brillant";
                message = "Superbe score ! Tu as une excellente culture générale.";
                emoji = "⭐";
            } else if (percentage >= 30) {
                badge = "🥈 Bon Apprenti";
                message = "Pas mal du tout ! Encore un petit effort et tu seras le champion.";
                emoji = "🚀";
            }

            document.getElementById('badgeContainer').textContent = `Badge : ${badge}`;
            document.getElementById('finalMessage').textContent = message;
            document.getElementById('resultEmoji').textContent = emoji;

            speak(`Partie terminée ! Votre score final est de ${score} points. ${message}`);

            if (typeof confetti === 'function') {
                confetti({
                    particleCount: 150,
                    spread: 100,
                    origin: { y: 0.6 }
                });
            }
        }

        function restartGame() {
            playSound('click');
            startGame();
        }
    </script>
